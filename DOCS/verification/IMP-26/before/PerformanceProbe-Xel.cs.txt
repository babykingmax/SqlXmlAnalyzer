using System.Collections.Concurrent;
using System.ComponentModel;
using System.Diagnostics;
using System.IO;
using System.Security.Cryptography;
using System.Text.Json;
using System.Windows;
using System.Windows.Controls;
using System.Windows.Media;
using System.Windows.Threading;
using System.Xml.Linq;
using Microsoft.Extensions.Logging.Abstractions;
using SqlXmlAnalyzer;
using SqlXmlAnalyzer.Analysis;
using SqlXmlAnalyzer.Application;
using SqlXmlAnalyzer.Application.Services;
using SqlXmlAnalyzer.Refactoring;
using SqlXmlAnalyzer.Services;
using SqlXmlAnalyzer.Views;

internal static class Program
{
    private static App CreateProbeApp()
    {
        var posted = new List<DispatcherOperation>();
        DispatcherHookEventHandler capture = (_, e) => posted.Add(e.Operation);
        var dispatcher = Dispatcher.CurrentDispatcher;
        dispatcher.Hooks.OperationPosted += capture;
        App app;
        try { app = new App(); }
        finally { dispatcher.Hooks.OperationPosted -= capture; }
        if (posted.Count != 1 || !posted[0].Abort()) throw new InvalidOperationException("Cannot isolate the queued Application startup callback.");
        return app;
    }
    [STAThread]
    private static int Main(string[] args)
    {
        App? app = null; MainWindow? main = null;
        var runs = new List<object>();
        try
        {
            AppContext.SetSwitch("Switch.System.Windows.Media.ShouldRenderEvenWhenNoDisplayDevicesAreAvailable", true);
            app = CreateProbeApp(); app.InitializeComponent();
            SynchronizationContext.SetSynchronizationContext(new DispatcherSynchronizationContext(Dispatcher.CurrentDispatcher));
            RenderOptions.ProcessRenderMode = System.Windows.Interop.RenderMode.SoftwareOnly;
            var files = new PhysicalFileHandler();
            var refactor = new SqlRefactoringEngine([], new DefaultRuleFilter(), NullLogger<SqlRefactoringEngine>.Instance);
            var orchestrator = new ApplicationOrchestrator(new SqlXmlAnalysisEngine(), refactor, files, new ConsoleResultReporter(), NullLogger<ApplicationOrchestrator>.Instance);
            var startup = Stopwatch.StartNew();
            main = new MainWindow(orchestrator, files) { Width = 1280, Height = 720, Left = -20000, Top = 0, WindowStartupLocation = WindowStartupLocation.Manual };
            main.Show(); main.Activate(); Pump();
            double startupMs = startup.Elapsed.TotalMilliseconds;
            string input = args[0];
            for (int run = 0; run < 2; run++)
            {
                var watch = Stopwatch.StartNew();
                var lag = new ConcurrentQueue<double>();
                var stages = new List<object>();
                var dispatcher = Dispatcher.CurrentDispatcher;
                var status = WorkspaceAccessibility.Descendants(main).OfType<ShellStatusBar>().Single().StatusTextBlock;
                var descriptor = DependencyPropertyDescriptor.FromProperty(TextBlock.TextProperty, typeof(TextBlock));
                EventHandler changed = (_, _) => stages.Add(new { Milliseconds = watch.Elapsed.TotalMilliseconds, Status = status.Text });
                descriptor.AddValueChanged(status, changed);
                using var timer = new System.Threading.Timer(_ =>
                {
                    long stamp = Stopwatch.GetTimestamp();
                    dispatcher.BeginInvoke(DispatcherPriority.Input, new Action(() => lag.Enqueue(Stopwatch.GetElapsedTime(stamp).TotalMilliseconds)));
                }, null, 0, 20);
                // Baseline XEL: omit its native Partial notification, retaining the same reader and selection pipeline.
                var opened = new SqlXmlAnalyzer.Core.Services.DocumentOpenService().OpenAsync(input);
                Until(() => opened.IsCompleted);
                var content = opened.GetAwaiter().GetResult();
                if (!content.HasUsableContent) throw new InvalidDataException(content.ErrorMessage);
                main.ViewModel.CurrentDeadlockAnalysis = null;
                main.ViewModel.CurrentDeadlockInput = content.Input;
                var field = typeof(MainWindow).GetField("_xelDeadlockUiActionService", System.Reflection.BindingFlags.Instance | System.Reflection.BindingFlags.NonPublic)!;
                var service = field.GetValue(main)!;
                service.GetType().GetMethod("ShowDocumentEvents")!.Invoke(service, [content.Input, input]);
                if (Path.GetExtension(input).Equals(".xel", StringComparison.OrdinalIgnoreCase))
                    Until(() => main.ViewModel.CurrentDeadlockAnalysis != null);
                main.UpdateLayout(); Pump();
                timer.Change(Timeout.Infinite, Timeout.Infinite); Pump();
                double total = watch.Elapsed.TotalMilliseconds;
                descriptor.RemoveValueChanged(status, changed);
                var sorted = lag.OrderBy(value => value).ToArray();
                var graph = WorkspaceAccessibility.Descendants(main).OfType<PlanWorkspaceView>().FirstOrDefault()?.NodifyGraph;
                runs.Add(new { Run = run == 0 ? "process-cold" : "same-process-warm", TotalMs = total, StartupMs = startupMs,
                    InputLatencyP95Ms = sorted.Length == 0 ? 0 : sorted[Math.Min(sorted.Length - 1, (int)Math.Ceiling(sorted.Length * .95) - 1)],
                    InputLatencyMaxMs = sorted.LastOrDefault(), Samples = sorted.Length,
                    PeakWorkingSetBytes = Process.GetCurrentProcess().PeakWorkingSet64, ManagedBytes = GC.GetTotalMemory(false),
                    VisibleGraphNodes = graph?.Nodes.Count ?? 0,
                    Operators = main.ViewModel.PlanWorkspace.Selection?.Operators.Count ?? 0,
                    XelEvents = main.ViewModel.CurrentDeadlockInput?.Deadlocks.Count ?? 0, Stages = stages });
                File.WriteAllText(args[1], JsonSerializer.Serialize(new { Passed = true, SourceSha256 = Convert.ToHexString(SHA256.HashData(File.ReadAllBytes(input))), Runs = runs }, new JsonSerializerOptions { WriteIndented = true }));
            }
            return 0;
        }
        catch (Exception exception)
        {
            File.WriteAllText(args[1], JsonSerializer.Serialize(new { Passed = false, Error = exception.ToString(), Runs = runs }, new JsonSerializerOptions { WriteIndented = true }));
            return 1;
        }
        finally { main?.Close(); app?.Shutdown(); }
    }
    private static void Pump() => Dispatcher.CurrentDispatcher.Invoke(() => { }, DispatcherPriority.ContextIdle);
    private static void Until(Func<bool> predicate)
    {
        var watch = Stopwatch.StartNew();
        while (!predicate()) { if (watch.Elapsed > TimeSpan.FromMinutes(3)) throw new TimeoutException("Performance probe timed out"); Pump(); Thread.Sleep(5); }
        Pump();
    }
}
