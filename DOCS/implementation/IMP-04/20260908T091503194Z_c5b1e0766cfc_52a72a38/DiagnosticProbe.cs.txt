using System.Security.Cryptography;
using System.Text.Json;
using Microsoft.Extensions.Logging;
using Microsoft.SqlServer.TransactSql.ScriptDom;
using SqlXmlAnalyzer.Application;
using SqlXmlAnalyzer.Application.Services;
using SqlXmlAnalyzer.Core;
using SqlXmlAnalyzer.Core.Abstractions;
using SqlXmlAnalyzer.Core.Diagnostics;
using SqlXmlAnalyzer.Refactoring;
using AppLogger = SqlXmlAnalyzer.Logger;

Console.OutputEncoding = new System.Text.UTF8Encoding(false);
string evidence = Path.GetFullPath(args[0]);
Directory.CreateDirectory(evidence);
AppLogger.Initialize(forceVerbose: true, logLevel: SqlXmlAnalyzer.LogLevel.Debug,
    customLogFilePath: Path.Combine(evidence, "runtime.log"));
AppLogger.Debug("probe-debug-marker");
AppLogger.Warning("probe-warning-marker");
AppLogger.Error("probe-error-marker");
AppLogger.Critical("probe-fatal-marker");
using var factory = LoggerFactory.Create(builder => builder.SetMinimumLevel(LogLevel.Trace).AddProvider(new ApplicationLoggerProvider()));
var diagnostics = new UnexpectedErrorReporter(new WindowsMiniDumpWriter(), Path.Combine(evidence, "dumps"));
var engine = new SqlRefactoringEngine(new ISqlRefactorRule[] { new SyntheticUnknownRule() }, new DefaultRuleFilter(),
    factory.CreateLogger<SqlRefactoringEngine>(), diagnostics);
string sqlPath = Path.Combine(evidence, "source.sql");
File.WriteAllText(sqlPath, "SELECT 1;");
string before = Convert.ToHexString(SHA256.HashData(File.ReadAllBytes(sqlPath)));
var orchestrator = new ApplicationOrchestrator(new EmptyAnalysis(), engine, new PhysicalFileHandler(),
    new JsonResultReporter(), factory.CreateLogger<ApplicationOrchestrator>(), unexpectedErrors: diagnostics);
var result = orchestrator.Execute(sqlPath);
string after = Convert.ToHexString(SHA256.HashData(File.ReadAllBytes(sqlPath)));
string? dump = result.Result?.Diagnostic?.DumpPath;
uint signature = 0, streams = 0;
if (dump != null)
{
    using var reader = new BinaryReader(File.OpenRead(dump));
    signature = reader.ReadUInt32();
    reader.ReadUInt32();
    streams = reader.ReadUInt32();
}
var verification = new
{
    Configuration = AppLogger.IsDebugMode ? "Debug" : "Release",
    SourceBeforeSha256 = before, SourceAfterSha256 = after,
    IsSuccess = result.IsSuccess, SourceWritten = result.Writeback?.SourceWritten ?? false,
    Diagnostic = result.Result?.Diagnostic, DumpSignature = signature, DumpStreamCount = streams,
    Passed = !result.IsSuccess && before == after && dump != null && signature == 0x504D444D && streams > 0
};
File.WriteAllText(Path.Combine(evidence, "verification.json"), JsonSerializer.Serialize(verification, new JsonSerializerOptions { WriteIndented = true }));
AppLogger.Shutdown();
return verification.Passed ? 1 : 2; // 1 is the deliberately failed application operation; 2 is a failed probe.

sealed class SyntheticUnknownRule : ISqlRefactorRule
{
    public string RuleId => "TEST_UNKNOWN_EXCEPTION";
    public string Name => "Synthetic unknown fault";
    public string Description => Name;
    public int Priority => 1;
    public bool CanApply(TSqlFragment fragment, RefactorContext context) => true;
    public RuleResult Apply(TSqlFragment fragment, RefactorContext context) => throw new InvalidOperationException("IMP-04 synthetic unknown error");
}
sealed class EmptyAnalysis : IAnalysisEngine
{
    public AnalysisReport Analyze(string content) => new(Array.Empty<IAnalysisIssue>());
}
